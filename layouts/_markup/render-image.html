{{- /* Absolute image URLs so images survive in feeds and email; width/height on raster bundle images to avoid layout shift. */ -}} {{- $u := urls.Parse .Destination }} {{- $src := .Destination }} {{- $w := 0 }}{{ $h := 0 }} {{- if not $u.IsAbs }} {{- with .PageInner.Resources.Get $u.Path }} {{- $src = .Permalink }} {{- if and (eq .ResourceType "image") (ne .MediaType.SubType "svg") }}{{ $w = .Width }}{{ $h = .Height }}{{ end }} {{- else }} {{- if hasPrefix $u.Path "/" }}{{ $src = absURL $u.Path }}{{ else }}{{ $src = urls.JoinPath .PageInner.Permalink $u.Path }}{{ end }} {{- end }} {{- end -}} {{- /**/ -}}
